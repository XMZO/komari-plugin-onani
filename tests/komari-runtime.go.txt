// Exercise the real plugin scheduler and cache against deterministic Komari RPC fixtures.
func TestOnaniHostnameRuntime(t *testing.T) {
	oldVersion := utils.CurrentVersion
	utils.CurrentVersion = "1.4.3"
	defer func() { utils.CurrentVersion = oldVersion }()
	withTempDataDir(t)
	gin.SetMode(gin.TestMode)
	Init(gin.New())
	root := os.Getenv("ONANI_PROJECT_ROOT")
	files := map[string]string{}
	for _, name := range []string{"komari-plugin.json", "script.js", "pages/index.html"} {
		data, err := os.ReadFile(filepath.Join(root, name))
		if err != nil {
			t.Fatal(err)
		}
		files[name] = string(data)
	}
	if _, err := InstallZip(writePluginZip(t, files)); err != nil {
		t.Fatal(err)
	}
	if err := SaveConfiguration("onani", map[string]any{"hostname_enabled": true, "hostname_auto_refresh": false}); err != nil {
		t.Fatal(err)
	}
	const uuid = "f9e79538-e4a8-43b2-8e58-a1857c82cc5f"
	var scenario atomic.Int32
	register := func(name string, handler rpc.Handler) {
		t.Helper()
		if err := rpc.Register(name, handler); err != nil {
			t.Fatal(err)
		}
		t.Cleanup(func() { rpc.Unregister(name) })
	}
	register("common:getNodes", func(context.Context, *rpc.JsonRpcRequest) (any, *rpc.JsonRpcError) {
		return map[string]any{uuid: map[string]any{"name": "diagnostic fixture"}}, nil
	})
	register("common:getNodesLatestStatus", func(context.Context, *rpc.JsonRpcRequest) (any, *rpc.JsonRpcError) {
		return map[string]any{uuid: map[string]any{"online": true}}, nil
	})
	var commandsMu sync.Mutex
	var commands []string
	register("admin:exec", func(_ context.Context, req *rpc.JsonRpcRequest) (any, *rpc.JsonRpcError) {
		var params struct {
			Command string `json:"command"`
		}
		_ = req.BindParams(&params)
		commandsMu.Lock()
		commands = append(commands, params.Command)
		commandsMu.Unlock()
		if scenario.Load() == 3 {
			return nil, &rpc.JsonRpcError{Code: -32001, Message: "dispatch fixture rejected"}
		}
		if params.Command == "uname -n" {
			return map[string]any{"task_id": "fallback-task-456", "clients": []string{uuid}}, nil
		}
		return map[string]any{"task_id": "diagnostic-task-123", "clients": []string{uuid}}, nil
	})
	register("admin:getTaskResultsByTaskId", func(_ context.Context, req *rpc.JsonRpcRequest) (any, *rpc.JsonRpcError) {
		var params struct {
			TaskID string `json:"task_id"`
		}
		_ = req.BindParams(&params)
		switch scenario.Load() {
		case 7, 8:
			// BusyBox without the hostname applet: sh reports 127, uname -n works unless scenario 8.
			if params.TaskID != "fallback-task-456" {
				return []any{map[string]any{"client": uuid, "exit_code": 127, "result": "sh: hostname: not found"}}, nil
			}
			if scenario.Load() == 8 {
				return []any{map[string]any{"client": uuid, "exit_code": 127, "result": "sh: uname: not found"}}, nil
			}
			return []any{map[string]any{"client": uuid, "exit_code": 0, "result": "fixture-busybox\n"}}, nil
		case 1:
			return []any{map[string]any{"client": uuid, "exit_code": -1, "result": "Remote control is disabled."}}, nil
		case 2:
			return []any{map[string]any{"client": uuid, "exit_code": 0, "result": ""}}, nil
		case 4:
			return nil, &rpc.JsonRpcError{Code: -32002, Message: "results fixture unavailable"}
		case 5:
			return []any{}, nil
		default:
			return []any{map[string]any{"client": uuid, "exit_code": 0, "result": "fixture-host"}}, nil
		}
	})
	if err := SetEnabled("onani", true, true); err != nil {
		t.Fatal(err)
	}
	defer SetEnabled("onani", false, false)
	ctx := rpc.NewContextWithMeta(context.Background(), &rpc.ContextMeta{Permission: rpc.RoleAdmin})
	call := func(method string, params any) map[string]any {
		t.Helper()
		res := rpc.CallWithContext(ctx, "test", method, params)
		if res.Error != nil {
			t.Fatalf("%s: %+v", method, res.Error)
		}
		raw, _ := json.Marshal(res.Result)
		var out map[string]any
		if err := json.Unmarshal(raw, &out); err != nil {
			t.Fatal(err)
		}
		return out
	}
	cases := []struct {
		id                    int32
		name, contains, stage string
		rpcCode, exitCode     int
		taskID, output        string
		commands              string
	}{
		{1, "agent failure", "Remote control is disabled.", "Agent 执行 hostname", 0, -1, "diagnostic-task-123", "Remote control is disabled.", "hostname"},
		{2, "empty hostname", "空主机名", "解析 hostname 输出", 0, 0, "diagnostic-task-123", "", "hostname"},
		{3, "dispatch failure", "dispatch fixture rejected", "下发 hostname 命令", -32001, 0, "", "", "hostname"},
		{4, "query failure", "results fixture unavailable", "查询远程任务结果", -32002, 0, "diagnostic-task-123", "", "hostname"},
		{5, "agent timeout", "尚未收到结果", "等待 Agent 返回结果", 0, 0, "diagnostic-task-123", "", "hostname"},
		{6, "recovery", "", "", 0, 0, "", "", "hostname"},
		{7, "busybox fallback", "", "", 0, 0, "", "", "hostname,uname -n"},
		{8, "fallback failure", "系统没有 hostname 命令；改用 uname -n 后，Agent 返回退出码 127", "Agent 执行 uname -n", 0, 127, "fallback-task-456", "sh: uname: not found", "hostname,uname -n"},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			scenario.Store(tc.id)
			commandsMu.Lock()
			commands = nil
			commandsMu.Unlock()
			call("plugin:onani.hostname.refresh", map[string]any{"force": true, "uuids": []string{uuid}})
			deadline := time.Now().Add(35 * time.Second)
			var state map[string]any
			for {
				state = call("plugin:onani.hostname.status", nil)
				if state["refresh"].(map[string]any)["running"] == false {
					break
				}
				if time.Now().After(deadline) {
					t.Fatal("refresh never completed")
				}
				time.Sleep(25 * time.Millisecond)
			}
			entry := state["entries"].(map[string]any)[uuid].(map[string]any)
			commandsMu.Lock()
			sent := strings.Join(commands, ",")
			commandsMu.Unlock()
			if sent != tc.commands {
				t.Fatalf("commands sent=%q want %q", sent, tc.commands)
			}
			if tc.id == 6 || tc.id == 7 {
				want := map[int32]string{6: "fixture-host", 7: "fixture-busybox"}[tc.id]
				if entry["hostname"] != want || entry["last_failure"] != nil || entry["last_error"] != nil {
					t.Fatalf("recovery: %+v", entry)
				}
				return
			}
			if !strings.Contains(entry["last_error"].(string), tc.contains) {
				t.Fatalf("wrong error: %+v", entry)
			}
			details, ok := entry["last_failure"].(map[string]any)
			if !ok || details["stage"] != tc.stage || details["at"] == nil {
				t.Fatalf("missing diagnostics: %+v", entry)
			}
			if tc.taskID != "" && details["task_id"] != tc.taskID {
				t.Fatalf("missing task: %+v", details)
			}
			if tc.rpcCode != 0 && details["rpc_code"] != float64(tc.rpcCode) {
				t.Fatalf("missing RPC code: %+v", details)
			}
			if (tc.id <= 2 || tc.id == 8) && details["exit_code"] != float64(tc.exitCode) {
				t.Fatalf("missing exit code: %+v", details)
			}
			if tc.output != "" && details["output"] != tc.output {
				t.Fatalf("missing output: %+v", details)
			}
			if tc.id == 8 && entry["hostname"] != "fixture-busybox" {
				t.Fatalf("failed fallback dropped the last good hostname: %+v", entry)
			}
			raw, err := os.ReadFile(filepath.Join(StorageDir, "onani", "hostname-cache.json"))
			if err != nil || !bytes.Contains(raw, []byte(tc.contains)) {
				t.Fatalf("diagnostics not persisted: %v %s", err, raw)
			}
			if !strings.Contains(GetLogs("onani"), tc.contains) {
				t.Fatal("diagnostics missing from plugin log")
			}
			refresh := state["refresh"].(map[string]any)
			if refresh["active_jobs"] != float64(0) || refresh["queued_jobs"] != float64(0) {
				t.Fatal("job slot leaked")
			}
			t.Logf("diagnostic: %s", entry["last_error"])
		})
	}
}

// Replay komari-zig-agent's legacy result upload through the real hook chain, Komari's
// agent authentication, the v2 JSON-RPC handler and the task database.
func TestOnaniAgentCompatRuntime(t *testing.T) {
	oldVersion := utils.CurrentVersion
	utils.CurrentVersion = "1.4.3"
	defer func() { utils.CurrentVersion = oldVersion }()
	withTempDataDir(t)
	gin.SetMode(gin.TestMode)
	engine := gin.New()
	Init(engine)
	// Same middleware order as the production router for agent endpoints.
	engine.Use(api.IdentityMiddleware())
	engine.Group("/api/clients", api.RequireRole(api.RoleAdmin, api.RoleClient)).POST("/v2/rpc", client.UploadV2RPC)
	handler := WrapHandler(engine)

	root := os.Getenv("ONANI_PROJECT_ROOT")
	files := map[string]string{}
	for _, name := range []string{"komari-plugin.json", "script.js", "pages/index.html"} {
		data, err := os.ReadFile(filepath.Join(root, name))
		if err != nil {
			t.Fatal(err)
		}
		files[name] = string(data)
	}
	if _, err := InstallZip(writePluginZip(t, files)); err != nil {
		t.Fatal(err)
	}
	if err := SaveConfiguration("onani", map[string]any{"hostname_enabled": false}); err != nil {
		t.Fatal(err)
	}
	if err := SetEnabled("onani", true, true); err != nil {
		t.Fatal(err)
	}
	defer SetEnabled("onani", false, false)

	uuid, token, err := clients.CreateClient()
	if err != nil {
		t.Fatal(err)
	}
	newTask := func(id string) {
		t.Helper()
		if err := tasks.CreateTask(id, []string{uuid}, "hostname"); err != nil {
			t.Fatal(err)
		}
	}
	post := func(rawQuery, body string, header map[string]string) *httptest.ResponseRecorder {
		r := httptest.NewRecorder()
		req := httptest.NewRequest(http.MethodPost, "/api/clients/task/result?"+rawQuery, strings.NewReader(body))
		req.Header.Set("Content-Type", "application/json")
		req.Header.Set("User-Agent", "komari-zig-agent")
		for name, value := range header {
			req.Header.Set(name, value)
		}
		handler.ServeHTTP(r, req)
		return r
	}
	stored := func(id string) (string, *int, *time.Time) {
		t.Helper()
		result, err := tasks.GetSpecificTaskResult(id, uuid)
		if err != nil {
			t.Fatal(err)
		}
		return result.Result, result.ExitCode, result.FinishedAt
	}
	query := "token=" + url.QueryEscape(token)

	t.Run("zig payload is stored through v2", func(t *testing.T) {
		newTask("compat-ok")
		body := `{"task_id":"compat-ok","result":"compat-host\n","exit_code":0,"finished_at":"2026-10-05T12:48:35Z"}`
		r := post(query, body, nil)
		if r.Code != http.StatusOK || !strings.Contains(r.Body.String(), "success") {
			t.Fatalf("legacy upload: %d %s\nlogs: %s", r.Code, r.Body.String(), GetLogs("onani"))
		}
		result, exitCode, finishedAt := stored("compat-ok")
		if result != "compat-host\n" || exitCode == nil || *exitCode != 0 {
			t.Fatalf("stored result=%q exit=%v", result, exitCode)
		}
		if finishedAt == nil || !finishedAt.Equal(time.Date(2026, 10, 5, 12, 48, 35, 0, time.UTC)) {
			t.Fatalf("finished_at=%v", finishedAt)
		}
	})

	t.Run("failure exit code and stale encoding header", func(t *testing.T) {
		newTask("compat-fail")
		body := `{"task_id":"compat-fail","result":"Remote control is disabled.","exit_code":-1,"finished_at":"not a time"}`
		r := post(query, body, map[string]string{"Content-Encoding": "gzip"})
		if r.Code != http.StatusOK {
			t.Fatalf("legacy failure upload: %d %s", r.Code, r.Body.String())
		}
		result, exitCode, finishedAt := stored("compat-fail")
		if result != "Remote control is disabled." || exitCode == nil || *exitCode != -1 || finishedAt == nil {
			t.Fatalf("stored result=%q exit=%v finished=%v", result, exitCode, finishedAt)
		}
	})

	t.Run("komari still authenticates the agent", func(t *testing.T) {
		newTask("compat-auth")
		body := `{"task_id":"compat-auth","result":"forged","exit_code":0}`
		if r := post("token=wrong-token", body, nil); r.Code != http.StatusUnauthorized {
			t.Fatalf("wrong token: %d %s", r.Code, r.Body.String())
		}
		if result, exitCode, _ := stored("compat-auth"); result != "" || exitCode != nil {
			t.Fatalf("unauthenticated result stored: %q %v", result, exitCode)
		}
	})

	t.Run("malformed bodies keep the original 404", func(t *testing.T) {
		if r := post(query, "not json", nil); r.Code != http.StatusNotFound {
			t.Fatalf("malformed body: %d %s", r.Code, r.Body.String())
		}
		if !strings.Contains(GetLogs("onani"), "旧版 Agent 任务回传未转换") {
			t.Fatalf("missing warning: %s", GetLogs("onani"))
		}
	})

	t.Run("oversized bodies are refused before the hook runs", func(t *testing.T) {
		body := `{"task_id":"compat-big","result":"` + strings.Repeat("x", 8<<20) + `","exit_code":0}`
		if r := post(query, body, nil); r.Code != http.StatusRequestEntityTooLarge {
			t.Fatalf("oversized body: %d", r.Code)
		}
	})

	t.Run("disabling the plugin removes the hook", func(t *testing.T) {
		newTask("compat-disabled")
		if err := SetEnabled("onani", false, false); err != nil {
			t.Fatal(err)
		}
		body := `{"task_id":"compat-disabled","result":"late","exit_code":0}`
		if r := post(query, body, nil); r.Code != http.StatusNotFound {
			t.Fatalf("disabled plugin: %d %s", r.Code, r.Body.String())
		}
	})
}

// The background proxy was removed in 0.3.0. Its leftover image cache must be deleted
// on load without touching the hostname cache that shares the storage directory.
func TestOnaniRetiredStorageRuntime(t *testing.T) {
	oldVersion := utils.CurrentVersion
	utils.CurrentVersion = "1.4.3"
	defer func() { utils.CurrentVersion = oldVersion }()
	withTempDataDir(t)
	gin.SetMode(gin.TestMode)
	Init(gin.New())
	root := os.Getenv("ONANI_PROJECT_ROOT")
	files := map[string]string{}
	for _, name := range []string{"komari-plugin.json", "script.js", "pages/index.html"} {
		data, err := os.ReadFile(filepath.Join(root, name))
		if err != nil {
			t.Fatal(err)
		}
		files[name] = string(data)
	}
	if _, err := InstallZip(writePluginZip(t, files)); err != nil {
		t.Fatal(err)
	}
	if err := SaveConfiguration("onani", map[string]any{"hostname_enabled": false}); err != nil {
		t.Fatal(err)
	}
	storage := filepath.Join(StorageDir, "onani")
	image := filepath.Join(storage, "background-cache", strings.Repeat("a", 64))
	if err := os.MkdirAll(image, 0o700); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(image, "original"), []byte("retired image"), 0o600); err != nil {
		t.Fatal(err)
	}
	hostnameCache := filepath.Join(storage, "hostname-cache.json")
	cacheBody := []byte(`{"schema":1,"updated_at":null,"entries":{}}`)
	if err := os.WriteFile(hostnameCache, cacheBody, 0o600); err != nil {
		t.Fatal(err)
	}
	if err := SetEnabled("onani", true, true); err != nil {
		t.Fatal(err)
	}
	defer SetEnabled("onani", false, false)

	deadline := time.Now().Add(5 * time.Second)
	for {
		if _, err := os.Stat(filepath.Join(storage, "background-cache")); os.IsNotExist(err) {
			break
		}
		if time.Now().After(deadline) {
			t.Fatalf("retired background cache was not removed\nlogs: %s", GetLogs("onani"))
		}
		time.Sleep(25 * time.Millisecond)
	}
	if data, err := os.ReadFile(hostnameCache); err != nil || !bytes.Equal(data, cacheBody) {
		t.Fatalf("hostname cache must survive the cleanup: %v %s", err, data)
	}
	if !strings.Contains(GetLogs("onani"), "removed retired storage: background-cache") {
		t.Fatalf("cleanup not logged: %s", GetLogs("onani"))
	}
}
